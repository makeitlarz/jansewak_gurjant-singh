var ss = SpreadsheetApp.getActiveSpreadsheet(); // Sheet names check or create helper function getOrCreateSheet(sheetName) { var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } return sheet; } function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var responseObj = { result: "success" }; if (action === "getData") { // 1. Issues & Complaints Data var issueSheet = getOrCreateSheet("Issues"); var issueRows = issueSheet.getDataRange().getValues(); var issues = []; for (var i = 1; i < issueRows.length; i++) { var row = issueRows[i]; if (row[0]) { // ID exists issues.push({ id: String(row[0]), name: row[1], phone: row[2], type: row[3], category: row[4], details: row[5], date: row[6], status: row[7] || "Pending", approved: String(row[8]) === "true" || row[8] === true, reply: row[9] || "" }); } } responseObj.issues = issues; // 2. Settings & Media Data (Banner, Profile, Work Gallery) var settingsSheet = getOrCreateSheet("Settings"); var settingsRows = settingsSheet.getDataRange().getValues(); var settings = {}; for (var j = 1; j < settingsRows.length; j++) { if (settingsRows[j][0]) { settings[settingsRows[j][0]] = settingsRows[j][1]; } } responseObj.settings = settings; } var jsonString = JSON.stringify(responseObj); if (callback) { return ContentService.createTextOutput(callback + "(" + jsonString + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { return handleRequest(e); } // JSONP / GET & POST दोनों के लिए हैंडलर function handleRequest(e) { var params = e.parameter; var action = params.action; var callback = params.callback; if (action === "update") { var issueSheet = getOrCreateSheet("Issues"); var id = String(params.id); var rows = issueSheet.getDataRange().getValues(); var found = false; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === id) { // Update existing row if (params.status !== undefined) issueSheet.getRange(i + 1, 8).setValue(params.status); if (params.approved !== undefined) issueSheet.getRange(i + 1, 9).setValue(params.approved); if (params.reply !== undefined) issueSheet.getRange(i + 1, 10).setValue(params.reply); found = true; break; } } if (!found) { // Append new issue (Headers: ID, Name, Phone, Type, Category, Details, Date, Status, Approved, Reply) if(issueSheet.getLastRow() === 0) { issueSheet.appendRow(["ID", "Name", "Phone", "Type", "Category", "Details", "Date", "Status", "Approved", "Reply"]); } issueSheet.appendRow([ id, params.name || "", params.phone || "", params.type || "समस्या", params.category || "", params.details || "", params.date || "", params.status || "Pending", params.approved || "false", params.reply || "" ]); } } else if (action === "saveSettings") { var settingsSheet = getOrCreateSheet("Settings"); if(settingsSheet.getLastRow() === 0) { settingsSheet.appendRow(["Key", "Value"]); } var key = params.key; var value = params.value; var rows = settingsSheet.getDataRange().getValues(); var updated = false; for (var k = 1; k < rows.length; k++) { if (rows[k][0] === key) { settingsSheet.getRange(k + 1, 2).setValue(value); updated = true; break; } } if (!updated) { settingsSheet.appendRow([key, value]); } } var output = JSON.stringify({ result: "success" }); if (callback) { return ContentService.createTextOutput(callback + "(" + output + ");") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(output) .setMimeType(ContentService.MimeType.JSON); } }
